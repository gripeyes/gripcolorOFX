cmake_minimum_required(VERSION 3.20)
project(Rendition VERSION 0.3.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(RENDITION_BUILD_OFX "Build macOS OFX bundle" ON)
option(RENDITION_BUILD_PYTHON "Build research evaluator" ON)
add_library(rendition_core src/core/core.cpp src/core/operators.cpp src/core/kernel_bridge.cpp src/core/diagnostics.cpp src/core/primaries.cpp src/core/artist_models.cpp)
target_include_directories(rendition_core PUBLIC include)
target_compile_options(rendition_core PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math -ffp-contract=off)
if(RENDITION_BUILD_PYTHON)
 find_package(Python COMPONENTS Interpreter Development.Module REQUIRED)
 execute_process(COMMAND ${Python_EXECUTABLE} -m pybind11 --cmakedir OUTPUT_VARIABLE PYBIND_DIR OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
 list(APPEND CMAKE_PREFIX_PATH ${PYBIND_DIR})
 find_package(pybind11 CONFIG REQUIRED)
 pybind11_add_module(_rendition tools/bindings.cpp)
 target_link_libraries(_rendition PRIVATE rendition_core)
endif()
include(CTest)
if(BUILD_TESTING AND RENDITION_BUILD_PYTHON)
 add_test(NAME independent_core COMMAND ${Python_EXECUTABLE} -m pytest ${CMAKE_SOURCE_DIR}/tests/test_core.py ${CMAKE_SOURCE_DIR}/tests/test_nuke_bridge.py ${CMAKE_SOURCE_DIR}/tests/test_diagnostics.py ${CMAKE_SOURCE_DIR}/tests/test_primaries.py ${CMAKE_SOURCE_DIR}/tests/test_artist_architecture.py -q)
 set_tests_properties(independent_core PROPERTIES ENVIRONMENT "PYTHONPATH=$<TARGET_FILE_DIR:_rendition>:${CMAKE_SOURCE_DIR}")
endif()
if(RENDITION_BUILD_OFX AND APPLE)
 add_library(Rendition MODULE src/ofx/plugin.cpp)
 target_include_directories(Rendition PRIVATE third_party/openfx)
 target_link_libraries(Rendition PRIVATE rendition_core)
 set_target_properties(Rendition PROPERTIES PREFIX "" SUFFIX ".ofx" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/Rendition.ofx.bundle/Contents/MacOS" CXX_VISIBILITY_PRESET hidden)
 file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/Rendition.ofx.bundle/Contents")
 configure_file(resources/Info.plist "${CMAKE_BINARY_DIR}/Rendition.ofx.bundle/Contents/Info.plist" @ONLY)
 add_custom_target(rendition_bundle_sign ALL
  COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_BINARY_DIR}/Rendition.ofx.bundle/Contents/Resources"
  COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_SOURCE_DIR}/third_party/openfx/LICENSE.md" "${CMAKE_BINARY_DIR}/Rendition.ofx.bundle/Contents/Resources/OpenFX-LICENSE.md"
  COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_SOURCE_DIR}/third_party/licenses/Colour-LICENSE" "${CMAKE_BINARY_DIR}/Rendition.ofx.bundle/Contents/Resources/Colour-LICENSE"
  COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_SOURCE_DIR}/THIRD_PARTY_REFERENCES.md" "${CMAKE_BINARY_DIR}/Rendition.ofx.bundle/Contents/Resources/THIRD_PARTY_REFERENCES.md"
  COMMAND /usr/bin/codesign --force --sign - "$<TARGET_FILE_DIR:Rendition>/../.."
  DEPENDS Rendition VERBATIM)
endif()
if(BUILD_TESTING)
 add_executable(rendition_native_tests tests/native.cpp)
 target_link_libraries(rendition_native_tests PRIVATE rendition_core)
 target_compile_options(rendition_native_tests PRIVATE -UNDEBUG)
 add_test(NAME native_image_contract COMMAND rendition_native_tests)
endif()
if(BUILD_TESTING AND RENDITION_BUILD_PYTHON)
 add_test(NAME operator_contract COMMAND ${Python_EXECUTABLE} -m pytest ${CMAKE_SOURCE_DIR}/tests/test_operators.py ${CMAKE_SOURCE_DIR}/tests/test_spectral.py ${CMAKE_SOURCE_DIR}/tests/test_full_reference.py ${CMAKE_SOURCE_DIR}/tests/test_spectral_completion.py ${CMAKE_SOURCE_DIR}/tests/test_restored_controls.py -q)
 set_tests_properties(operator_contract PROPERTIES ENVIRONMENT "PYTHONPATH=$<TARGET_FILE_DIR:_rendition>:${CMAKE_SOURCE_DIR}")
endif()
option(RENDITION_BUILD_METAL_VALIDATOR "Build shared-equation offline Metal parity validator" ON)
if(APPLE AND RENDITION_BUILD_METAL_VALIDATOR)
 enable_language(OBJCXX)
 set(METAL_LIBRARY "${CMAKE_BINARY_DIR}/Rendition.metallib")
 add_custom_command(OUTPUT "${METAL_LIBRARY}"
  COMMAND xcrun metal -std=macos-metal2.4 -fno-fast-math -ffp-contract=off -I "${CMAKE_SOURCE_DIR}/include" -c "${CMAKE_SOURCE_DIR}/src/metal/kernels.metal" -o "${CMAKE_BINARY_DIR}/Rendition.air"
  COMMAND xcrun metallib "${CMAKE_BINARY_DIR}/Rendition.air" -o "${METAL_LIBRARY}"
  DEPENDS src/metal/kernels.metal include/rendition/kernel_math.hpp include/rendition/kernel_parameters.hpp VERBATIM)
 add_custom_target(rendition_metal_library ALL DEPENDS "${METAL_LIBRARY}")
 add_executable(rendition_metal_validate tools/metal_validate.mm)
 target_link_libraries(rendition_metal_validate PRIVATE rendition_core "-framework Metal" "-framework Foundation")
 target_compile_options(rendition_metal_validate PRIVATE -fobjc-arc -fno-fast-math -ffp-contract=off)
 add_dependencies(rendition_metal_validate rendition_metal_library)
 if(BUILD_TESTING)
  add_test(NAME metal_parity COMMAND rendition_metal_validate "${METAL_LIBRARY}" "${CMAKE_BINARY_DIR}/metal-parity.json")
 endif()
endif()
if(APPLE AND RENDITION_BUILD_METAL_VALIDATOR)
 add_executable(rendition_metal_benchmark tools/benchmark_metal.mm)
 target_link_libraries(rendition_metal_benchmark PRIVATE rendition_core "-framework Metal" "-framework Foundation")
 target_compile_options(rendition_metal_benchmark PRIVATE -fobjc-arc -fno-fast-math -ffp-contract=off)
 add_dependencies(rendition_metal_benchmark rendition_metal_library)
endif()
